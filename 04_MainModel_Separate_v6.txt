"""
Bayesian_RealData_Separate_v7.py
Separate mortality model: predation mortality of age-0 (P0) and overwinter
mortality of age-0 (W0) as distinct components.
Free parameters: beta0, W1, P0.
Same likelihood structure as the combined model: Poisson for m2,
lambda1 anchored to m1 with +/-5% counting uncertainty (u),
lambda0 = m0/N0 exact.
v7: demographic group totals added as pm.Deterministic for exact
    posterior summaries.
Output: BayesianSeparate_Results_Summary_v7.xlsx
"""

import pandas as pd
import numpy as np
import pymc as pm
import arviz as az
from scipy.optimize import differential_evolution
import warnings
warnings.filterwarnings('ignore')

np.random.seed(42)

U_MIN = 0.95
U_MAX = 1.05

# ============================================================
# 1. READ DATA
# ============================================================

df = pd.read_excel('Separate_M.xlsx', header=0)
df = df.set_index('Parameters')

params = {}
for morph in ['Nosed', 'White']:
    params[morph] = {
        'm0': df.loc['lambda_Age 0+', morph],
        'm1': df.loc['lambda_Age 1+', morph],
        'm2': df.loc['lambda_Age 2+', morph],
        'P_min': df.loc['P₀_min', morph],
        'P_max': df.loc['P₀_max', morph],
        'W0_min': df.loc['W₀_min', morph],
        'W0_max': df.loc['W₀_max', morph],
        'W1_min': df.loc['W₁_min', morph],
        'W1_max': df.loc['W₁_max', morph],
        'n_spawn_min': df.loc['n_spawners_min', morph],
        'n_spawn_max': df.loc['n_spawners_max', morph],
        'fec_min': df.loc['fecundity_min', morph],
        'fec_max': df.loc['fecundity_max', morph],
        'egg_mort_min': df.loc['egg_mortality_min', morph],
        'egg_mort_max': df.loc['egg_mortality_max', morph],
    }

for morph in ['Nosed', 'White']:
    p = params[morph]
    n_spawn = (p['n_spawn_min'] + p['n_spawn_max']) / 2
    fec = (p['fec_min'] + p['fec_max']) / 2
    egg_surv = 1 - (p['egg_mort_min'] + p['egg_mort_max']) / 2
    p['N0_base'] = n_spawn * fec * egg_surv
    p['lambda0'] = p['m0'] / p['N0_base']

# ============================================================
# 2. FEASIBLE BOUNDS (common box for both morphotypes)
# ============================================================

def level1_objective(bounds, params, delta=0.05):
    M_min, M_max, W1_min, W1_max = bounds
    if M_min >= M_max or W1_min >= W1_max or M_min <= W1_max + delta:
        return 1e10
    total_obj = 0
    for morph in ['Nosed', 'White']:
        p = params[morph]
        lambda0 = p['lambda0']
        beta0_min = max(0.001, 1 - M_max - lambda0)
        beta0_max = 1 - M_min - lambda0
        if beta0_min >= beta0_max:
            total_obj += 1e6
            continue
        beta0_test = np.linspace(beta0_min, beta0_max, 1000)
        lambda1_vals = p['m1'] / (p['N0_base'] * beta0_test)
        beta1_vals = p['m2'] / (p['N0_base'] * beta0_test)
        W1_vals = 1 - beta1_vals - lambda1_vals
        M0_vals = 1 - beta0_test - lambda0
        feasible = (W1_vals >= W1_min) & (W1_vals <= W1_max) & (M0_vals > W1_vals + delta)
        for i in range(len(beta0_test)):
            if not feasible[i]:
                continue
            M0_i = M0_vals[i]
            P0_min_feas = max(p['P_min'], M0_i - p['W0_max'])
            P0_max_feas = min(p['P_max'], M0_i - p['W0_min'])
            if P0_min_feas > P0_max_feas:
                feasible[i] = False
        if not np.any(feasible):
            total_obj += 1e6
            continue
    return total_obj

result_level1 = differential_evolution(
    level1_objective, [(0.05, 0.5), (0.3, 0.95), (0.05, 0.5), (0.05, 0.5)],
    args=(params, 0.05), seed=42, maxiter=300, tol=1e-8, polish=True
)
M_min_opt, M_max_opt, W1_min_opt, W1_max_opt = result_level1.x

print(f"Feasible bounds: M0=[{M_min_opt:.3f}, {M_max_opt:.3f}], W1=[{W1_min_opt:.3f}, {W1_max_opt:.3f}]")

# ============================================================
# 3. BAYESIAN MODEL (3 free parameters: beta0, W1, P0)
# ============================================================

all_summaries = {}
SUM_VARS = ['beta0', 'W1', 'P0', 'W0', 'M0', 'beta1', 'lambda1',
            'N_b0_Tt', 'N_P0_Tt', 'N_W0_Tt', 'N_M0_Tt',
            'N_W1_Tt', 'N_b1_Tt', 'N_l1_Tt']

for morph in ['Nosed', 'White']:
    p = params[morph]
    N0_fixed = p['N0_base']
    lambda0 = p['lambda0']

    beta0_min = max(0.001, 1 - M_max_opt - lambda0)
    beta0_max = 1 - M_min_opt - lambda0

    P0_min_overall = max(p['P_min'], 1 - beta0_max - lambda0 - p['W0_max'])
    P0_max_overall = min(p['P_max'], 1 - beta0_min - lambda0 - p['W0_min'])
    if P0_min_overall >= P0_max_overall:
        P0_min_overall, P0_max_overall = p['P_min'], p['P_max']

    print(f"\n=== {morph} === beta0:[{beta0_min:.4f},{beta0_max:.4f}] "
          f"W1:[{W1_min_opt:.4f},{W1_max_opt:.4f}] P0:[{P0_min_overall:.4f},{P0_max_overall:.4f}]")

    with pm.Model() as model:
        beta0 = pm.Uniform('beta0', lower=beta0_min, upper=beta0_max)
        W1 = pm.Uniform('W1', lower=W1_min_opt, upper=W1_max_opt)
        P0 = pm.Uniform('P0', lower=P0_min_overall, upper=P0_max_overall)
        u = pm.Uniform('u_m1', lower=U_MIN, upper=U_MAX)

        W0 = pm.Deterministic('W0', 1 - beta0 - lambda0 - P0)
        M0 = pm.Deterministic('M0', P0 + W0)
        lambda1 = pm.Deterministic('lambda1', (p['m1'] / (N0_fixed * beta0)) * u)
        beta1 = pm.Deterministic('beta1', 1 - W1 - lambda1)

        # Demographic group totals (exact posterior summaries)
        N_b0_Tt = pm.Deterministic('N_b0_Tt', N0_fixed * beta0)
        N_P0_Tt = pm.Deterministic('N_P0_Tt', N0_fixed * P0)
        N_W0_Tt = pm.Deterministic('N_W0_Tt', N0_fixed * W0)
        N_M0_Tt = pm.Deterministic('N_M0_Tt', N0_fixed * M0)
        N_W1_Tt = pm.Deterministic('N_W1_Tt', N0_fixed * beta0 * W1)
        N_b1_Tt = pm.Deterministic('N_b1_Tt', N0_fixed * beta0 * beta1)
        N_l1_Tt = pm.Deterministic('N_l1_Tt', N0_fixed * beta0 * lambda1)

        mu_m2 = N0_fixed * beta0 * beta1
        pm.Poisson('likelihood_m2', mu=mu_m2, observed=p['m2'])

        pm.Potential('M0_gt_W1',
                     -10000 * pm.math.switch(M0 > W1 + 0.05, 0, (W1 + 0.05 - M0)**2))
        pm.Potential('W0_positive',
                     -10000 * pm.math.switch(W0 > 0.001, 0, (0.001 - W0)**2))
        pm.Potential('W0_in_bounds',
                     -10000 * (pm.math.switch(W0 >= p['W0_min'], 0, (p['W0_min'] - W0)**2) +
                               pm.math.switch(W0 <= p['W0_max'], 0, (W0 - p['W0_max'])**2)))

        trace = pm.sample(1500, tune=1500, chains=2, cores=2,
                         return_inferencedata=True,
                         target_accept=0.92,
                         progressbar=False,
                         random_seed=42)

    summary = az.summary(trace, var_names=SUM_VARS, hdi_prob=0.95)
    all_summaries[morph] = summary

    print(f"\n--- Posterior Summary ---")
    print(summary.to_string())
    rhat_max = summary['r_hat'].max()
    ess_min = summary['ess_bulk'].min()
    divergences = int(trace.sample_stats.diverging.sum().values)
    print(f"Diagnostics: R-hat(max)={rhat_max:.3f}, ESS(min)={ess_min:.0f}, Divergences={divergences}")

# ============================================================
# 4. SUMMARY TABLE
# ============================================================

rows = []
for morph in ['Nosed', 'White']:
    summary = all_summaries[morph]
    for param in ['beta0', 'P0', 'W0', 'M0', 'N_b0_Tt', 'N_P0_Tt', 'N_W0_Tt', 'N_M0_Tt',
                  'W1', 'beta1', 'lambda1', 'N_W1_Tt', 'N_b1_Tt', 'N_l1_Tt']:
        row = summary.loc[param]
        rows.append({'Morph': morph, 'Parameter': param,
                     'Mean': row['mean'], 'SD': row['sd'],
                     'HDI_2.5%': row['hdi_2.5%'], 'HDI_97.5%': row['hdi_97.5%'],
                     'ESS_bulk': row['ess_bulk'], 'R_hat': row['r_hat']})

df_results = pd.DataFrame(rows)
df_results.to_excel('BayesianSeparate_Results_Summary_v7.xlsx', index=False)
print("\nSaved: BayesianSeparate_Results_Summary_v7.xlsx")
